cmake_minimum_required(VERSION 3.24)
project(RocketChad3 LANGUAGES CXX)
option(ROCKETCHAD_BUILD_NATIVE "" OFF)
set(ROCKETGPU_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../RocketGPU" CACHE PATH "")
if(ROCKETCHAD_BUILD_NATIVE)
    enable_language(CUDA)
    set(ROCKETGPU_USE_CUDA ON CACHE BOOL "" FORCE)
    set(ROCKETGPU_BUILD_PYTHON ON CACHE BOOL "" FORCE)
    add_subdirectory("${ROCKETGPU_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/rocketgpu")
    find_package(CUDAToolkit REQUIRED)
    find_package(Python COMPONENTS Interpreter Development REQUIRED)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_rocketgpu_start_cache native/start_cache.cpp native/start_cache.cu)
    target_include_directories(_rocketgpu_start_cache PRIVATE "${ROCKETGPU_SOURCE_DIR}/include")
    target_link_libraries(_rocketgpu_start_cache PRIVATE rocketgpu CUDA::cudart)
    target_compile_features(_rocketgpu_start_cache PRIVATE cxx_std_17)
    target_compile_options(_rocketgpu_start_cache PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()
